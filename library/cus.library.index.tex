\cus_hook_gput_pe:n
  {
    \DeclareDocumentCommand \makeindex { +O{} O{} } { \newindextype [{#1}] {#2} }
    \DeclareDocumentCommand \index { >{\TrimSpaces} O{} +m }
      {
        \cus_index_if_exist:nTF {#1}
          {
            \@bsphack
            \group_begin:
            \cs_set_eq:Nc \@indexfile { @indexfile #1 }
            \@sanitize
            \@wrindex {#2}
          }
          { \msg_error:nnn { cus } { index-unknown } {#1} }
      }
    \DeclareDocumentCommand \printindex { +O{} }
      {
        \group_begin:
        \tl_clear:N \l__cus_tmp_tl
        \tl_clear:N \l__cus_tmpa_tl
        \keyval_parse:NNn \__cus_index_check_name:n \__cus_index_check_name:nn {#1}
        \str_if_empty:NT \l__cus_tmp_tl
          { \tl_set_eq:NN \l__cus_tmp_tl \c__cus_index_blank_tl }
        \seq_if_in:NVTF \g__cus_index_seq \l__cus_tmp_tl
          {
            \use:e
              {
                \EditInstance { cus/index }
                  { \exp_not:o \l__cus_tmp_tl }
                  { \exp_not:o \l__cus_tmpa_tl }
              }
            \tl_set:Ne \l__cus_index_title_tl
              { \__cus_index_value:nn { \l__cus_tmp_tl } { title } }
            \tl_set:Ne \l__cus_index_multi_tl
              { \__cus_index_value:nn { \l__cus_tmp_tl } { multi } }
            \tl_set:Ne \l__cus_index_heading_tl
              { \__cus_index_value:nn { \l__cus_tmp_tl } { heading } }
            \tl_set:Ne \l__cus_index_heading_cmd_tl
              { \__cus_index_value:nn { \l__cus_tmp_tl } { heading* } }
            \tl_set:Ne \l__cus_index_init_tl
              { \__cus_index_value:nn { \l__cus_tmp_tl } { init } }
            \tl_set:Ne \l__cus_index_prologue_tl
              { \__cus_index_value:nn { \l__cus_tmp_tl } { prologue } }
            \tl_set:Ne \l__cus_index_heading_multi_tl
              {
                \tl_if_empty:NTF \l__cus_index_title_tl { }
                  {
                    \tl_if_empty:NTF \l__cus_index_heading_tl
                      {
                        \exp_not:o \l__cus_index_heading_tl
                        { \exp_not:o \l__cus_index_title_tl }
                      }
                      {
                        \exp_not:o \l__cus_index_heading_cmd_tl
                        { \exp_not:o \l__cus_index_title_tl }
                      }
                  }
              }
            \tl_set:Ne \l__cus_index_empty_do_tl
              {
                \exp_not:o \l__cus_index_heading_multi_tl
                \exp_not:c { @index@ \l__cus_tmp_tl @prologue }
              }
            \cs_set:Npx \theindex
              {
                \cus@theindex
                  {
                    heading =
                      {
                        \exp_not:o \l__cus_index_heading_multi_tl
                        \exp_not:c { @index@ \l__cus_tmp_tl @prologue }
                      } ,
                    \exp_not:o \l__cus_index_multi_tl
                  }
                \index@presetting { \l__cus_tmp_tl }
                \par 
              }
            \cs_set_eq:NN \endtheindex \endcus@theindex
            \cs:w @index@ \l__cus_tmp_tl @init \cs_end:
            \@input@ { \__cus_index_value:nn { \l__cus_tmp_tl } { output } }
            \l__cus_index_empty_do_tl
          }
          { \msg_error:nnn { cus } { index-unknown } {#1} }
        \group_end:
      }
  }
